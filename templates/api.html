Using the API from Python 3

Base URL: {{request.host_url}}api/v1. Send the token as Authorization: Bearer <token>. Responses are JSON and errors look like {"error": "..."}. List endpoints accept limit (max 500) and offset. Install the client library with python3 -m pip install requests.

import os
import requests

BASE = "{{request.host_url}}api/v1"
session = requests.Session()
session.headers["Authorization"] = "Bearer " + os.environ["MARE_TOKEN"]
# session.verify = "/path/to/ca.pem"   # for an internal certificate authority

def check(response):
    if not response.ok:
        raise RuntimeError(f"{response.status_code}: {response.text}")
    return response

# Who am I, and what may I do?
print(check(session.get(f"{BASE}/me")).json())

# Create a case and read it back
case = check(session.post(f"{BASE}/cases", json={"name": "20260101: Example sample"})).json()["case"]
case_id = case["id"]

# Assign an owner and start analysis (required before editing content)
me = session.get(f"{BASE}/me").json()["user"]
check(session.patch(f"{BASE}/cases/{case_id}", json={"owner_id": me["id"]}))
check(session.post(f"{BASE}/cases/{case_id}/stage", json={"expected": 0}))

# Add a task and a note
check(session.post(f"{BASE}/cases/{case_id}/tasks", json={"title": "Unpack the sample"}))
check(session.post(f"{BASE}/cases/{case_id}/notes", json={"title": "Triage", "body": "Packed with UPX."}))

# Upload a file (multipart) into a package category such as scripts or signatures
with open("decode.py", "rb") as handle:
    check(session.post(f"{BASE}/cases/{case_id}/assets",
                       data={"category": "scripts", "description": "Config decoder"},
                       files={"file": handle}))

# Upload a pasted screenshot for a report section
with open("screenshot.png", "rb") as handle:
    image = check(session.post(f"{BASE}/cases/{case_id}/report/images",
                               data={"section": "key_findings"},
                               files={"image": handle})).json()
print("Use in Markdown: ![caption](" + image["markdown_url"] + ")")

# Upload and select your Word template once (needs a heading or placeholder per report section)
with open("template.docx", "rb") as handle:
    check(session.post(f"{BASE}/templates", files={"template": handle}))

# Save report sections, then generate the Word report and wait for the job
check(session.put(f"{BASE}/cases/{case_id}/report", json={"sections": {"executive_summary": "Summary text"}}))
check(session.post(f"{BASE}/cases/{case_id}/report/generate", json={"previous": "overwrite"}))
import time
while True:
    job = session.get(f"{BASE}/cases/{case_id}/report/job").json()["job"]
    if job and job["status"] != "running" and job["status"] != "queued":
        break
    time.sleep(2)

# Download the generated report, then upload an edited copy
with open("report.docx", "wb") as out:
    out.write(check(session.get(f"{BASE}/cases/{case_id}/report/download")).content)
with open("report-edited.docx", "rb") as handle:
    check(session.post(f"{BASE}/cases/{case_id}/report/final", files={"report": handle}))

# Export the MIP archive (Packaging & Delivery or Completed stage)
archive = check(session.post(f"{BASE}/cases/{case_id}/archive", json={"archive_type": "standard"}))
with open("mip.zip", "wb") as out:
    out.write(archive.content)

# Back up every case
with open("all-cases.zip", "wb") as out:
    out.write(check(session.post(f"{BASE}/backups/export")).content)

Case-changing calls follow the same rules as the web interface: an active owner is required, and content cannot be edited while a case is Not started or Completed. Status codes: 400 invalid input, 401 missing or invalid token, 403 missing permission, 404 not found, 409 conflict with the case state, 429 rate limit exceeded (600 requests per minute per token).
